@inherits ExampleBase
@using System.Diagnostics
@using MudBlazor.Extensions.Office.Document

<MudStack Row="true" AlignItems="AlignItems.Center" Spacing="4" Class="mb-3">
    <MudButton Variant="Variant.Filled" Color="Color.Primary" OnClick="@Generate">@L["Generate a large document"]</MudButton>
    <MudText Typo="Typo.body2">@_info</MudText>
</MudStack>

@if (_document != null)
{
    <div style="width: 100%; height: 520px;">
        <MudExDocumentEditor @ref="ComponentRef" @bind-TextDocument="_document" />
    </div>
}

@code {
    private TextDocument? _document;
    private string _info = "";

    private async Task Generate()
    {
        _document = null;
        _info = L["Generating..."];
        await Task.Yield();
        var watch = Stopwatch.StartNew();
        _document = DocumentSamples.Large();
        watch.Stop();
        _info = L["{0} paragraphs in {1:N0} ms", _document.AllParagraphs().Count(), watch.ElapsedMilliseconds];
    }
}
